1) ""أَلَا حَيِّ رَهبَى ثم حَيِّ المَطاليا    فقد كان مأنوسًا فأَصبح خاليا""
2) ""فلا عَهْدَ إِلا أَن تَذَكَّرَ أَو تَرَى    ثُمَامًا حَوالَيْ مَنصِب الخَيْم باليا""
3) ""إِلى اللهِ أَشكو أَنَّ بالغَوْرِ حاجةً    وأُخرى إِذا أَبصرت نجدًا بَدَا ليا""
4) ""إِذا ما أَراد الحي أَن يتزايلوا    وحنت جِمال البين حنت جماليا""
5) ""أَلا أَيها الوادي الذي ضم سيله    إِلينا نوى ظمياءَ حُيِّيتَ واديا""
6) ""تَخَطَّى إِلينا من بعيد خيالُها    يخوضُ خُداريًّا من الليل داجيا""
7) ""نظرتُ بِرَهْبَى والظعائن باللِّوَى    فطارت برهبى شُعبة من فؤاديا""
8) ""فإِنكِ إِن تُعْطِي قليلا فطال ما    منعتِ وحلّأْت القلوب الصواديا""
9) ""دُنُوَّ عتاق الطير للزجر بعدما    شَمَسْنَ وولَّيْنَ الخدود العواصيا""
10) ""تُعَيِّرني الإِخلافَ ليلى وأَفْضَلَتْ    على وصل ليلى قُوَّةٌ من حباليا""
11) ""فما أَبصر النارَ التي وضحت له    وراءَ جُفاف الطير إِلا تماريا""
12) ""لَحِقْتُ وأَصحابي على كل حُرَّةٍ    مَرُوح تُبارِي الأَحْبَشِيَّ المُكارِيَا""
13) ""تَرامَيْن بالأَجواز في كل صَفْصَفٍ    فأدنين من خَلْجِ البُرين الذفاريا""
14) ""إِذا بلَّغت رحلي رجيعٌ أَملها    نزوليَ بِالمَوْماةِ ثم ارتحاليا""
15) ""مُخَفِّقة يهوِي على الهول رَكْبُها    قليلا بها ما ينظرون التواليا""
16) ""تجول بها موتى الشخاص كأَنها    قذى غَرَق يُضْحِي به الماء طافيا""
17) ""لشق على ذي الحلم أَن يتبع الهوى    ويرجو من الأَقصى الذي ليس لاقيا""
18) ""وكائن ترى في الحي من ذي صداقة    وغَيرانَ يدعُو وَيْلَه من حِذَاريا""
19) ""إِذا ذُكرت هند أتيح ليَ الهوى    على ما ترى من هِجرتي واجتنابيا""
20) ""خليليَّ لَوْلا أَن تظنا بِيَ الهوى    لقلت سمعنا من سُكينة داعيا""
21) ""قفا فاسمعا صوت المنادي فإِنه    قَرٍيبٌ وما دانيتُ بالوُد دَانيا""
22) ""أَلا طرقت أَسماءُ لا حِين مَطْرَق    أَحَمَّ عُمانيا وأَشعث ماضيا""
23) ""لدى قَطَريات إِذا ما تَغَوَّلَتْ    بنا البيد غاولن الحُزُوم القياقيا""
24) ""فَحُيِّيتَ مِنْ سَارٍ تَكَلَّفَ مَوْهِنًا    مَزارًا على ذي حاجة متراخيا""
25) ""إِذا ما جَعلتُ السِّيَّ بيني وبينها    وحرَّةَ ليلى والعَقِيقَ اليمانيا""
26) ""دعوتُ إِلى ذي العرش ربِّ محمد    لِيَجْمَعَ شعبًا أَو يُقَرِّبَ نائيًا""
27) ""ويأْمرني العُذال أَن أَغلب الهوى    وأَن أُخفِيَ الوجدَ الذي ليس خافيا""
28) ""فيا حسراتِ القلب في إِثْرِ من يُرى    قريبًا، ويُلفى خَيْرُهُ منك قاصيا""
29) ""تعرضتُ فاستمررتِ من دون حاجتي    فحالَكِ إِني مُسْتَمِرٌّ لحاليا""
30) ""فَرُدِّي جِمال البينِ ثم تحمّلي    فما لَكِ فيهم من مُقام ولا ليا""
31) ""وإِني لَمَغْرُورٌ أُعَلِّلُ بالمُنَى    لياليَ أَرجو أَن مالكَ ماليا""
32) ""وقائلةٍ والدمعُ يَحْدُرُ كُحْلَها    أَبعْدَ جرير تُكْرِمُونَ المواليا""
33) ""بِأَيِّ نجاد تحمل السيفَ بعدما    قَطَعْتَ القُوى من مِحْمَلٍ كان باقيا""
34) ""بأَي سِنان تطعُن القوم بعدما    نزعتَ سِنانًا من قناتك ماضيا""
35) ""أَلا لا تخافا نَبْوَتِي في مُلِمَّةٍ    وخافا المنايا أَن تفوتكما بيا""
36) ""فقد كنتُ نارًا يصطليها عدوُّكم    وحِرْزًا لما أَلْجَأْتُمُ من ورائيا""
37) ""وباسِطَ خير فيكمُ بيمينه    وقابِضَ شر عنكمُ بشماليا""
38) ""وإِني لَعَفُّ الفقر مُشْتَرَكُ الغِنَى    سَرِيعٌ -إِذا لم أَرض داري- انتقاليا""
39) ""جَرِيُّ الجَنان لا أُهال من الردى    إِذا ما جعلت السيفَ من عن شماليا""
40) ""وليست لسيفي في العظام بقيةٌ    ولَلسَّيْفُ أَشوى وَقْعَةً من لسانيا""
